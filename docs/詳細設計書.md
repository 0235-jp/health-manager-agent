import { HealthData, CustomInstruction, ReportContent } from '../../types';

/**
 * Agent SDKの抽象インターフェース
 * 新しいAgent SDKを追加する場合はこのインターフェースを実装する
 */
export interface AgentAdapter {
  /**
   * アダプタ名（ログ・デバッグ用）
   */
  readonly name: string;

  /**
   * アダプタの初期化
   */
  initialize(): Promise<void>;

  /**
   * 評価レポートを生成
   */
  generateReport(params: GenerateReportParams): Promise<ReportContent>;

  /**
   * リソースの解放
   */
  dispose?(): Promise<void>;
}

export interface GenerateReportParams {
  /** レポートタイプ */
  reportType: 'on_fetch' | 'daily';
  /** 評価対象期間の開始 */
  periodStart: Date;
  /** 評価対象期間の終了 */
  periodEnd: Date;
  /** ヘルスデータ（事前取得済み、オプション） */
  healthData?: HealthData[];
  /** カスタム指示 */
  customInstructions?: CustomInstruction[];
}

export interface ReportContent {
  summary: string;
  metrics: Record<string, {
    value: number;
    unit: string;
    trend: 'up' | 'down' | 'stable';

  recommendations: string[];

#### 1.4.3 AgentService（Factoryパターン）

```typescript
// packages/server/src/agent/index.ts

import { AgentAdapter } from './interfaces/agent-adapter';
import { AnthropicAgentAdapter } from './adapters/anthropic-adapter';
import { settingsRepository } from '../db/repositories/settings';

export class AgentService {
  private static instance: AgentService;
  private adapter: AgentAdapter | null = null;

  private constructor() {}

  static getInstance(): AgentService {
    if (!AgentService.instance) {
      AgentService.instance = new AgentService();
    }
    return AgentService.instance;
  }

  async initialize(): Promise<void> {
    const settings = await settingsRepository.getAll();
    const adapterType = settings.agent_adapter || 'anthropic';

    this.adapter = this.createAdapter(adapterType);
    await this.adapter.initialize();
  }

  private createAdapter(type: string): AgentAdapter {
    switch (type) {
      case 'anthropic':
        return new AnthropicAgentAdapter();
      // 将来的に他のアダプタを追加可能
      // case 'openai':
      //   return new OpenAIAgentAdapter();
      // case 'langchain':
      //   return new LangChainAgentAdapter();
      default:
        throw new Error(`Unknown agent adapter type: ${type}`);
    }
  }

  getAdapter(): AgentAdapter {
    if (!this.adapter) {
      throw new Error('AgentService not initialized');
    }
    return this.adapter;
  }

  async generateReport(params: GenerateReportParams): Promise<ReportContent> {
    return this.getAdapter().generateReport(params);
  }
}
```

#### 1.4.4 Anthropic Agent SDK アダプタ実装

```typescript
// packages/server/src/agent/adapters/anthropic-adapter.ts

import { Agent } from '@anthropic-ai/claude-agent-sdk';
import { AgentAdapter, GenerateReportParams, ReportContent } from '../interfaces/agent-adapter';
import { getHealthDataSkill } from '../skills/get-health-data';
import { buildSystemPrompt } from '../prompts';

export class AnthropicAgentAdapter implements AgentAdapter {
  readonly name = 'anthropic';
  private agent: Agent | null = null;

  async initialize(): Promise<void> {
    this.agent = new Agent({
      model: 'claude-sonnet-4-20250514',
      tools: [getHealthDataSkill],
    });
  }

  async generateReport(params: GenerateReportParams): Promise<ReportContent> {
    if (!this.agent) {
      throw new Error('Agent not initialized');
    }

    const systemPrompt = buildSystemPrompt(
      params.reportType,
      params.customInstructions || []
    );

    const response = await this.agent.run({
      system: systemPrompt,
      messages: [
        {
          role: 'user',
          content: `${params.periodStart.toISOString()}から${params.periodEnd.toISOString()}までのヘルスデータを分析し、評価レポートを作成してください。`
        }
      ]
    });

    return this.parseReportContent(response);
  }

  private parseReportContent(response: unknown): ReportContent {
    // レスポンスからReportContent形式にパース
    // 実装詳細は省略
    return response as ReportContent;
  }

  async dispose(): Promise<void> {
    this.agent = null;
  }
}

// packages/server/src/agent/prompts.ts

import { CustomInstruction } from '../types';

export function buildSystemPrompt(
  reportType: 'on_fetch' | 'daily',
  customInstructions: CustomInstruction[]
): string {
  let prompt = `あなたはヘルスデータアナリストです。
ユーザーのヘルスデータを分析し、${reportType === 'daily' ? '日次' : '定期'}評価レポートを作成してください。

レポートには以下を含めてください：
- 全体的な健康状態のサマリー
- 各指標の現在値とトレンド
- リスクや注意点
- 改善のための推奨事項

レポートは以下のJSON形式で出力してください：
{
  "summary": "全体サマリー",
  "metrics": {
    "metric_name": { "value": 数値, "unit": "単位", "trend": "up|down|stable" }
  },
  "risks": ["リスク1", "リスク2"],
  "recommendations": ["推奨事項1", "推奨事項2"]
}
`;

  if (customInstructions.length > 0) {
    prompt += `\n\n## ユーザーからの特別な指示\n以下の点に特に注意してください：\n`;
    customInstructions
      .sort((a, b) => b.priority - a.priority)
      .forEach(inst => {
        prompt += `- ${inst.instruction}\n`;
      });
  }

  return prompt;
}
```

### 1.5 スケジューラ詳細

```typescript
// packages/server/src/scheduler/index.ts

import cron from 'node-cron';
import { PluginManager } from '../plugins/manager';
import { AgentService } from '../agent';
import { WebhookSender } from '../webhook/sender';
import { settingsRepository } from '../db/repositories/settings';

export class Scheduler {
  private dataCollectionJob: cron.ScheduledTask | null = null;
  private dailyReportJob: cron.ScheduledTask | null = null;
  private reminderJobs: Map<string, cron.ScheduledTask> = new Map();

  async start(): Promise<void> {
    const settings = await settingsRepository.getAll();

    // データ収集ジョブ
    const interval = settings.collection_interval || 3600;
    this.dataCollectionJob = cron.schedule(
      this.secondsToCron(interval),
      () => this.runDataCollection()
    );

    // 日次レポートジョブ（毎日0:05）
    this.dailyReportJob = cron.schedule('5 0 * * *', () => this.runDailyReport());

    // リマインダージョブ
    await this.setupReminderJobs();
  }

  private async runDataCollection(): Promise<void> {
    const pluginManager = PluginManager.getInstance();
    const agentService = AgentService.getInstance();
    const webhook = new WebhookSender();

    // 1. プラグインからデータ取得
    const data = await pluginManager.executeAll({
      startDate: this.getLastCollectionTime(),
      endDate: new Date()
    });

    // 2. DBに保存
    await healthDataRepository.bulkInsert(data);

    // 3. 取得時レポート生成（AgentServiceを通じて実行）
    const report = await agentService.generateReport({
      reportType: 'on_fetch',
      periodStart: this.getLastCollectionTime(),
      periodEnd: new Date()
    });

    await reportsRepository.insert(report);

    // 4. Webhook送信
    await webhook.sendReport(report);
  }

  private secondsToCron(seconds: number): string {
    if (seconds < 60) return `*/${seconds} * * * * *`;
    const minutes = Math.floor(seconds / 60);
    if (minutes < 60) return `*/${minutes} * * * *`;
    const hours = Math.floor(minutes / 60);
    return `0 */${hours} * * *`;
  }
}
```

## 3. 検証計画

### 3.1 単体テスト

| モジュール | テスト内容 |
|-----------|-----------|
| PluginLoader | ZIP展開、マニフェスト検証、エラーハンドリング |
| PluginManager | プラグイン読み込み、データ収集、競合解決 |
| HealthAgent | Skill実行、レポート生成、カスタム指示適用 |
| Scheduler | ジョブ登録、cron式生成、エラー時リトライ |
| API Routes | リクエスト検証、レスポンス形式、エラーレスポンス |

### 3.2 統合テスト

1. **データ収集フロー**
   - プラグインからデータ取得 → DB保存 → レポート生成 → Webhook送信

2. **プラグイン管理フロー**
   - ZIPアップロード → インストール → 設定 → 接続テスト → データ取得

3. **フロントエンド連携**
   - API呼び出し → データ表示 → CRUD操作

### 3.3 E2Eテスト

```bash
# サーバー起動
cd packages/server && npm run dev

# フロントエンド起動
cd packages/frontend && npm run dev

# テストシナリオ
1. ダッシュボードにアクセス
2. 手動でデータを追加
3. グラフに反映されることを確認
4. プラグインをインストール
5. 手動同期を実行
6. 取得データが表示されることを確認
7. レポートを生成
8. Webhook送信を確認
```
